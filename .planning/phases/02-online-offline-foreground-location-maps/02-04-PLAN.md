const BASE_URL = process.env.EXPO_PUBLIC_API_BASE_URL;

export async function requestWithBase<T>(
  baseUrl: string | undefined,
  path: string,
  options: RequestOptions = {},
): Promise<T> {
  // ... the exact current apiRequest body, with `${BASE_URL}${path}` replaced by
  // `${baseUrl}${path}`. Everything else is unchanged: the Content-Type header, the
  // conditional Authorization header from useSessionStore, the `status === 204`
